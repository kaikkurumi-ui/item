# 1.商品表结构

```sql
CREATE TABLE `item` (
    id BIGINT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY ,
    item_images varchar(2000) CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci NOT NULL COMMENT '商品图片，用$符拼接',
    title varchar(100) CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci NOT NULL COMMENT '商品标题',
    price decimal NOT NULL COMMENT '商品价格',
    description varchar(2000) CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci  NOT NULL  COMMENT '商品详情介绍',
    create_time int unsigned NOT NULL ,
    update_time int unsigned NOT NULL ,
    is_deleted tinyint unsigned NOT NULL DEFAULT 0,
    KEY `idx_name` (`title`)
)ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT '商品信息表'
```

&nbsp;

### 1.2修改商品表，新增分类id字段

```sql
ALTER TABLE item
ADD COLUMN category_id BIGINT UNSIGNED NOT NULL COMMENT '分类id' AFTER is_deleted,
ADD INDEX idx_category_id (category_id);
```

# 2.分类表结构

```sql
CREATE TABLE `category` (
    id BIGINT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY ,
    name VARCHAR(50) CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci NOT NULL COMMENT '分类名称',
    category_image VARCHAR(2000) CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci NOT NULL COMMENT '分类图片',
    description VARCHAR(2000) CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci NOT NULL COMMENT '分类详情描述',
    create_time int unsigned NOT NULL ,
    update_time int unsigned NOT NULL ,
    is_deleted tinyint unsigned NOT NULL DEFAULT 0
)ENGINE=InnoDB DEFAULT  CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='分类信息表';
```

# 3.文件表结构

```sql
CREATE TABLE `file` (
    id BIGINT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY ,
    type TINYINT UNSIGNED NOT NULL COMMENT '1-图⽚，2-视频，3-⽂件',
    resource_uri VARCHAR(2000) CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci NOT NULL COMMENT '资源存放地点',
    create_time INT UNSIGNED NOT NULL COMMENT '创建时间',
    update_time INT UNSIGNED NOT NULL COMMENT '更新时间',
    is_deleted TINYINT UNSIGNED NOT NULL DEFAULT 0 COMMENT '逻辑删除，默认值0'
)ENGINE=InnoDB DEFAULT  CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='文件存储表';
```

# 4.用户表结构

```sql
CREATE TABLE `user` (
    id BIGINT UNSIGNED NOT NULL AUTO_INCREMENT PRIMARY KEY ,
    name VARCHAR(50) CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci NOT NULL COMMENT '用户名字',
    password VARCHAR(200) CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci NOT NULL COMMENT '密码',
    salt VARCHAR(100) CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci NOT NULL COMMENT '盐',
    phone VARCHAR(200) CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci NOT NULL COMMENT '手机号',
    avatar VARCHAR(2000) CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci NOT NULL  COMMENT '用户头像',
    create_time INT UNSIGNED NOT NULL COMMENT '创建时间',
    update_time INT UNSIGNED NOT NULL COMMENT '更新时间',
    is_deleted TINYINT UNSIGNED NOT NULL DEFAULT 0 COMMENT '逻辑删除，默认值0',

    UNIQUE INDEX idx_phone (phone)
)ENGINE=InnoDB DEFAULT  CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='用户信息表';
```
